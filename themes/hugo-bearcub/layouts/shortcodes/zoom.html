{{- /* A single image that opens in the zoom viewer. Params: src, alt, caption (optional), full (optional larger version), text="true" keeps it the width of the text column instead of wider. If src matches a file in the assets folder, it is resized for display and the original opens in the viewer. Otherwise src is used as a plain URL. */ -}} {{- $src := .Get "src" -}} {{- $fullParam := .Get "full" -}} {{- $alt := .Get "alt" | default (.Get "caption") -}} {{- $thumb := $src -}} {{- $full := $fullParam | default $src -}} {{- with resources.Get (strings.TrimPrefix "/" $src) -}} {{- if gt .Width 900 -}} {{- $thumb = (.Resize "900x").RelPermalink -}} {{- else -}} {{- $thumb = .RelPermalink -}} {{- end -}} {{- if not $fullParam }}{{ $full = .RelPermalink }}{{ end -}} {{- end -}} {{- if not (.Page.Scratch.Get "zoom-viewer") }} {{- .Page.Scratch.Set "zoom-viewer" true }} {{- partial "zoom-viewer.html" . }} {{- end }}
